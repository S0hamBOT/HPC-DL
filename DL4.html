# STEP 1: IMPORTS
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import r2_score, mean_squared_error
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout

# STEP 2: LOAD & CLEAN
df = pd.read_csv('/content/googl_data_2020_2025.csv')
print(df.columns.tolist())

# Convert all columns to numeric (force conversion)
df = df.apply(pd.to_numeric, errors='coerce')

# Drop columns that are completely NaN
df = df.dropna(axis=1, how='all')

# Pick close/open column
possible_cols = ['close', 'adj close', 'price', 'last', 'open']
col = None
for pc in possible_cols:
    for c in df.columns:
        if pc == c.lower():
            col = c
            break
    if col:
        break

# fallback safely
if col is None:
    if len(df.columns) == 0:
        raise ValueError("No usable numeric columns found in dataset")
    col = df.columns[-1]

print("Using column:", col)

# Drop NaN rows in that column
df = df[[col]].dropna()

# STEP 3: EDA
plt.figure(figsize=(10,4))
plt.plot(df[col].values)
plt.title('Stock Price History')
plt.xlabel('Days'); plt.ylabel('Price')
plt.grid(True); plt.show()

# STEP 4: SPLIT + SCALE + SEQUENCES
split = int(len(df) * 0.8)
train_df, test_df = df[:split], df[split:]

scaler = MinMaxScaler()
train_scaled = scaler.fit_transform(train_df[[col]].values)

SEQ = 60
X_train, y_train = [], []
for i in range(SEQ, len(train_scaled)):
    X_train.append(train_scaled[i-SEQ:i, 0])
    y_train.append(train_scaled[i, 0])

X_train = np.array(X_train).reshape(-1, SEQ, 1)
y_train = np.array(y_train)
print('X_train:', X_train.shape, '| y_train:', y_train.shape)

# STEP 5: BUILD MODEL
model = Sequential([
    LSTM(50, return_sequences=True, input_shape=(SEQ, 1)), Dropout(0.2),
    LSTM(50, return_sequences=True),                       Dropout(0.2),
    LSTM(50, return_sequences=True),                       Dropout(0.2),
    LSTM(50),                                              Dropout(0.2),
    Dense(1)
])
model.compile(optimizer='adam', loss='mean_squared_error')
model.summary()

# STEP 6: TRAIN
history = model.fit(X_train, y_train, epochs=50, batch_size=32)
plt.plot(history.history['loss'])
plt.title('Training Loss'); plt.xlabel('Epoch'); plt.ylabel('MSE')
plt.grid(True); plt.show()

# STEP 7: TEST DATA PREP + PREDICT
full   = df[col].values.reshape(-1, 1)
inputs = scaler.transform(full[len(full)-len(test_df)-SEQ:])

X_test = np.array([inputs[i-SEQ:i, 0] for i in range(SEQ, SEQ+len(test_df))]).reshape(-1, SEQ, 1)

predicted = scaler.inverse_transform(model.predict(X_test))
real      = test_df[[col]].values

# STEP 8: VISUALISE + EVALUATE
plt.figure(figsize=(10,5))
plt.plot(real,      label='Real Price',      color='blue')
plt.plot(predicted, label='Predicted Price', color='red')
plt.title('Stock Price Prediction — RNN (LSTM)')
plt.xlabel('Trading Day'); plt.ylabel('Price')
plt.legend(); plt.grid(True); plt.show()

r2   = r2_score(real, predicted)
rmse = np.sqrt(mean_squared_error(real, predicted))
print(f'R² Score : {r2:.4f}')
print(f'RMSE     : {rmse:.4f}')
